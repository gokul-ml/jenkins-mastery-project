pipeline {
    agent any

    environment {
        APP_VERSION = '2.0.0'
        ENV_TYPE    = 'Staging-Environment'
    }

    stages {
        stage('Environment Check') {
            steps {
                echo "Deploying Version ${env.APP_VERSION} to ${env.ENV_TYPE}."
                bat 'python --version'
            }
        }
        
        stage('Run Unit Tests') {
            steps {
                echo 'Executing automated testing suite...'
                bat 'python test.py'
            }
        }

        stage('Deploy Application') {
            steps {
                echo 'Pushing artifacts to live environment...'
                bat 'python app.py'
            }
        }
    }

    post {
        always {
            echo 'Pipeline run completed. Cleaning up workspace directories...'
        }
        success {
            echo 'CONGRATULATIONS: Build passed all quality checks perfectly!'
        }
        failure {
            echo 'WARNING: Build failed. Notification sent to development team.'
        }
    }
}
